      function shouldHideMostWantedSection(data) {
        if (!data || data.success === false) {
          return true;
        }

        const enabledFlag = data.config?.enabled ?? data.section_enabled ?? data.config?.most_wanted_enabled;

        if (typeof enabledFlag === 'string') {
          const normalized = enabledFlag.trim().toLowerCase();
          if (['true', '1', 'yes'].includes(normalized)) {
            return false;
          }
          if (['false', '0', 'no'].includes(normalized)) {
            return true;
          }
        } else if (typeof enabledFlag === 'boolean') {
          return !enabledFlag;
        } else if (typeof enabledFlag === 'number') {
          return enabledFlag !== 1;
        }

        const message = typeof data.message === 'string' ? data.message.toLowerCase() : '';
        return message.includes('desabilitada');

      function animateCounter() {
        const counter = document.getElementById('counter');

        if (isNaN(target)) return; // Verificar se data-target é válido

        const duration = 2000;
        const startTime = performance.now();

        function updateCounter(currentTime) {
          const elapsed = currentTime - startTime;
          const progress = Math.min(elapsed / duration, 1);
          
          const easeOut = 1 - Math.pow(1 - progress, 3);
          const current = Math.floor(start + (target - start) * easeOut);
          
          if (counter) { // Verificar novamente antes de usar
            counter.textContent = current.toLocaleString('pt-BR');
          }
          
          if (progress < 1) {
            requestAnimationFrame(updateCounter);
          }
        }
        
        requestAnimationFrame(updateCounter);
      }

        const enabledValue = data.store_enabled;

        let disabled = [];
